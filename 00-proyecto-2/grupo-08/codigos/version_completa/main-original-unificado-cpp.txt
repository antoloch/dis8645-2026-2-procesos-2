// utilizado por DORE 8. NO es la biblioteca Adafruit original sin modificar.
// Backend: ST7789V fisico (por defecto) o ILI9341 en Wokwi.
// SPI0: SCK GP18, MOSI GP19, CS GP17, DC GP16; RST GP20 solo fisico.
// Para compilar modo Wokwi definir DORE8_PANTALLA_ILI9341.
// Sin Arduino, sin framebuffer, sin bitmaps de escenas.
// 03 · CLASE GRÁFICA: DECLARACIÓN
// Esta interfaz reúne las operaciones que necesitan los escenarios (texto, líneas y figuras).
// Su API recuerda a Adafruit GFX, pero la implementación es propia para Pico SDK.
class AdafruitGFX_Pico {
public:
    // El constructor recibe los GPIO del selector CS, datos/comandos DC y reinicio RST.
    AdafruitGFX_Pico(uint8_t cs = 17, uint8_t dc = 16, int8_t rst = 20);
    // Inicializa SPI, los pines y el controlador de pantalla.
    void begin();

    void setCursor(int16_t x, int16_t y) { cursorX_ = x; cursorY_ = y; }
    std::size_t print(const char *txt);

    void getTextBounds(const char *txt, int16_t x, int16_t y,

    // Primitivas de dibujo que forman paredes, vehículos, personajes y marcos.

// 05 · IMPLEMENTACIÓN DEL CONTROLADOR TFT
// Se guardan los GPIO que manejarán la comunicación con la pantalla.
AdafruitGFX_Pico::AdafruitGFX_Pico(uint8_t cs, uint8_t dc, int8_t rst)
    : cs_(cs), dc_(dc), rst_(rst) {}

// Envía un comando al controlador: DC en 0 y CS activo durante el envío SPI.
void AdafruitGFX_Pico::comando(uint8_t valor) {
    gpio_put(dc_, false);
    gpio_put(cs_, false);
    spi_write_blocking(spi0, &valor, 1);
    gpio_put(cs_, true);
}
// Envía datos asociados a un comando (coordenadas, colores o configuración).
void AdafruitGFX_Pico::datos(const uint8_t *buffer, std::size_t n) {
    if (!n) return;
    gpio_put(dc_, true);
    gpio_put(cs_, false);
    spi_write_blocking(spi0, buffer, n);
    gpio_put(cs_, true);
}
// Esta clase conserva las primitivas de dibujo ya usadas por DORE 8.
// Solo cambia la inicializacion/orientacion segun el controlador.
// NO utiliza las bibliotecas de Arduino.
// Inicialización del bus y del panel; el modo físico usa ST7789V y Wokwi puede usar ILI9341.
void AdafruitGFX_Pico::begin() {
    // 12 MHz es un punto de partida prudente con cables Dupont/protoboard.
    // Se puede aumentar tras verificar el funcionamiento fisico.
    // Se inicia SPI0 a 12 MHz, velocidad conservadora para cables del prototipo.
    spi_init(spi0, 12000000);
    // GP18 y GP19 se reservan para el reloj y los datos de pantalla.
    gpio_set_function(18, GPIO_FUNC_SPI); // SCK
    gpio_set_function(19, GPIO_FUNC_SPI); // MOSI
    // Se configuran CS y DC como salidas, sin seleccionar aún la pantalla.
    gpio_init(cs_); gpio_set_dir(cs_, GPIO_OUT); gpio_put(cs_, true);
    gpio_init(dc_); gpio_set_dir(dc_, GPIO_OUT); gpio_put(dc_, true);

// En hardware físico se usa RST (GP20) para reiniciar el controlador ST7789V.
#if !defined(DORE8_PANTALLA_ILI9341)
    // El GMT020-02-8P fisico tiene RST independiente, conectado a GP20.
    if (rst_ >= 0) {
        gpio_init(static_cast<uint>(rst_));
        gpio_set_dir(static_cast<uint>(rst_), GPIO_OUT);
        gpio_put(static_cast<uint>(rst_), true);
        sleep_ms(10);
        gpio_put(static_cast<uint>(rst_), false);
        sleep_ms(20);
        gpio_put(static_cast<uint>(rst_), true);
        sleep_ms(150);
    }
#endif

    // Después del reinicio físico (si corresponde), se reinicia el controlador por software.
    comando(0x01); sleep_ms(150); // SWRESET (ambos controladores).

// Si la compilación define DORE8_PANTALLA_ILI9341, se elige la secuencia de Wokwi.
#if defined(DORE8_PANTALLA_ILI9341)
    // Mantener comportamiento de la pantalla ILI9341 simulada en Wokwi.
    comando(0x28);               // DISPOFF
    comando(0x3A);               // COLMOD: RGB565
    const uint8_t formato = 0x55; datos(&formato, 1);
    setRotation(1);
    comando(0x11); sleep_ms(150);
    comando(0x29); sleep_ms(30);
#else
    // ST7789V del panel fisico GMT020-02-8P (240 x 320).
    // En la ruta física ST7789V, se sale del modo reposo antes de dibujar.
    comando(0x11); sleep_ms(120);  // SLPOUT
    comando(0x3A);                 // COLMOD: 16 bits por pixel
    const uint8_t formato = 0x55; datos(&formato, 1);
    comando(0x21);                 // INVON: suele ser necesario en ST7789
    comando(0x13);                 // NORON: modo normal
    setRotation(1);                // Horizontal: 320 x 240
    comando(0x29); sleep_ms(80);   // DISPON
#endif
}

// Configura la orientación mediante MADCTL y actualiza ancho/alto en memoria.
void AdafruitGFX_Pico::setRotation(uint8_t r) {
    rotacion_ = r & 3;
// Si la compilación define DORE8_PANTALLA_ILI9341, se elige la secuencia de Wokwi.
#if defined(DORE8_PANTALLA_ILI9341)
    constexpr uint8_t direcciones[4] = {0x48, 0x28, 0x88, 0xE8};
#else
    // ST7789: MY, MX, MV. Si la imagen aparece girada o reflejada,
    // ajustar esta tabla despues de la primera prueba fisica.
    constexpr uint8_t direcciones[4] = {0xC0, 0xA0, 0x00, 0x60};
#endif
    comando(0x36); datos(&direcciones[rotacion_], 1); // MADCTL
    ancho_ = (rotacion_ & 1) ? 320 : 240;
    alto_ = (rotacion_ & 1) ? 240 : 320;
}
// Delimita el rectángulo de píxeles que recibirá la pantalla mediante comandos SPI.
void AdafruitGFX_Pico::ventana(int16_t x0, int16_t y0, int16_t x1, int16_t y1) {
    comando(0x2A);
    const uint8_t xs[] = {uint8_t(x0 >> 8), uint8_t(x0), uint8_t(x1 >> 8), uint8_t(x1)};
    datos(xs, 4);
    comando(0x2B);
    const uint8_t ys[] = {uint8_t(y0 >> 8), uint8_t(y0), uint8_t(y1 >> 8), uint8_t(y1)};
    datos(ys, 4);
    comando(0x2C);
}
// Dibuja rectángulos rellenos; recorta lo que quede fuera de la pantalla.
void AdafruitGFX_Pico::fillRect(int16_t x, int16_t y, int16_t w, int16_t h,
                               uint16_t color) {
    if (w <= 0 || h <= 0 || x >= ancho_ || y >= alto_ || x + w <= 0 || y + h <= 0) return;
    int x0 = std::max<int>(0,x), y0 = std::max<int>(0,y);
    int x1 = std::min<int>(ancho_,x+w), y1 = std::min<int>(alto_,y+h);
    if (x1 <= x0 || y1 <= y0) return;
    ventana(x0,y0,x1-1,y1-1);
    // Se utiliza un bloque pequeño de 128 bytes en lugar de guardar la pantalla completa en RAM.
    uint8_t bloque[128];
    for (int i=0;i<128;i+=2) { bloque[i]=uint8_t(color>>8); bloque[i+1]=uint8_t(color); }
    uint32_t pixeles = static_cast<uint32_t>(x1-x0)*(y1-y0);
    gpio_put(dc_,true); gpio_put(cs_,false);
    while (pixeles) {
        uint32_t n=std::min<uint32_t>(64,pixeles);
        spi_write_blocking(spi0,bloque,n*2);
        pixeles-=n;
    }
    gpio_put(cs_,true);
}
// Limpia la pantalla llamando al dibujo de un rectángulo del tamaño completo.
void AdafruitGFX_Pico::fillScreen(uint16_t color) { fillRect(0,0,ancho_,alto_,color); }
void AdafruitGFX_Pico::drawPixel(int16_t x,int16_t y,uint16_t c) { fillRect(x,y,1,1,c); }
void AdafruitGFX_Pico::drawFastHLine(int16_t x,int16_t y,int16_t w,uint16_t c) { fillRect(x,y,w,1,c); }
void AdafruitGFX_Pico::drawRect(int16_t x,int16_t y,int16_t w,int16_t h,uint16_t c) {
    if (w <= 0 || h <= 0) return;
    fillRect(x,y,w,1,c);fillRect(x,y+h-1,w,1,c);
    fillRect(x,y,1,h,c);fillRect(x+w-1,y,1,h,c);
}
// Traza líneas; optimiza las horizontales/verticales y calcula los píxeles de diagonales.
void AdafruitGFX_Pico::drawLine(int16_t x0,int16_t y0,int16_t x1,int16_t y1,uint16_t c) {
    if(y0==y1){fillRect(std::min(x0,x1),y0,std::abs(x1-x0)+1,1,c);return;}
    if(x0==x1){fillRect(x0,std::min(y0,y1),1,std::abs(y1-y0)+1,c);return;}
    int dx=std::abs(x1-x0),sx=x0<x1?1:-1;
    int dy=-std::abs(y1-y0),sy=y0<y1?1:-1;
    int error=dx+dy;
    while(true){
        drawPixel(x0,y0,c);
        if(x0==x1 && y0==y1)break;
        const int doble=2*error;
        if(doble>=dy){error+=dy;x0+=sx;}
        if(doble<=dx){error+=dx;y0+=sy;}
    }
}
void AdafruitGFX_Pico::circuloPuntos(int16_t cx,int16_t cy,int16_t x,int16_t y,uint16_t c) {
    drawPixel(cx+x,cy+y,c);drawPixel(cx-x,cy+y,c);drawPixel(cx+x,cy-y,c);drawPixel(cx-x,cy-y,c);
    drawPixel(cx+y,cy+x,c);drawPixel(cx-y,cy+x,c);drawPixel(cx+y,cy-x,c);drawPixel(cx-y,cy-x,c);
}
// Dibuja círculos con simetría; se usan en cabezas y ruedas.
void AdafruitGFX_Pico::drawCircle(int16_t cx,int16_t cy,int16_t r,uint16_t c) {
    if(r<0)return;
    int x=0,y=r,d=1-r;
    while(x<=y){circuloPuntos(cx,cy,x,y,c);if(d<0)d+=2*x+3;else{d+=2*(x-y)+5;--y;}++x;}
}
// Rellena círculos para detalles sólidos sin emplear un bitmap.
void AdafruitGFX_Pico::fillCircle(int16_t cx,int16_t cy,int16_t r,uint16_t c) {
    if(r<0)return;
    int x=0,y=r,d=1-r;
    while(x<=y){
        fillRect(cx-x,cy+y,2*x+1,1,c);fillRect(cx-x,cy-y,2*x+1,1,c);
        fillRect(cx-y,cy+x,2*y+1,1,c);fillRect(cx-y,cy-x,2*y+1,1,c);
        if(d<0)d+=2*x+3;else{d+=2*(x-y)+5;--y;}++x;
    }
}
// Dibuja marcos con esquinas redondeadas como parte de la estética retro.
void AdafruitGFX_Pico::drawRoundRect(int16_t x,int16_t y,int16_t w,int16_t h,int16_t r,uint16_t c) {
    if(w<=0||h<=0)return;
    r=std::max<int>(0,std::min<int>(r,std::min(w,h)/2));
    drawFastHLine(x+r,y,w-2*r,c);drawFastHLine(x+r,y+h-1,w-2*r,c);
    fillRect(x,y+r,1,h-2*r,c);fillRect(x+w-1,y+r,1,h-2*r,c);
    for(int i=0;i<=r;i++){
        // Esquina usando circulo discretizado con radio r.
        int j=0;
        while(j<r && i*i+j*j<r*r)++j;
        drawPixel(x+r-j,y+r-i,c);drawPixel(x+w-r-1+j,y+r-i,c);
        drawPixel(x+r-j,y+h-r-1+i,c);drawPixel(x+w-r-1+j,y+h-r-1+i,c);
    }
}
// Rellena las carrocerías de los vehículos usando geometría.
void AdafruitGFX_Pico::fillRoundRect(int16_t x,int16_t y,int16_t w,int16_t h,int16_t r,uint16_t c) {
    if(w<=0||h<=0)return;
    r=std::max<int>(0,std::min<int>(r,std::min(w,h)/2));
    fillRect(x+r,y,w-2*r,h,c);
    for(int i=0;i<r;i++){
        int j=0;
        while(j<r && i*i+j*j<r*r)++j;
        int xx=r-j;
        fillRect(x+xx,y+r-i,w-2*xx,1,c);
        fillRect(x+xx,y+h-r-1+i,w-2*xx,1,c);
    }
}
// Rasteriza triángulos mediante franjas horizontales, útil para flechas e indicadores.
void AdafruitGFX_Pico::fillTriangle(int16_t x0,int16_t y0,int16_t x1,int16_t y1,
                                   int16_t x2,int16_t y2,uint16_t c) {
    const int minY=std::min<int>({y0,y1,y2}),maxY=std::max<int>({y0,y1,y2});
    const int xs[]={x0,x1,x2},ys[]={y0,y1,y2};
    for(int y=minY;y<=maxY;++y){
        int interceptos[6],num=0;
        for(int i=0;i<3;++i){
            const int j=(i+1)%3;
            if(ys[i]==ys[j])continue;
            if(y>=std::min(ys[i],ys[j]) && y<=std::max(ys[i],ys[j]))
                interceptos[num++]=xs[i]+(y-ys[i])*(xs[j]-xs[i])/(ys[j]-ys[i]);
        }
        if(num>=2){
            int a=interceptos[0],b=interceptos[1];
            for(int k=2;k<num;k++){a=std::min(a,interceptos[k]);b=std::max(b,interceptos[k]);}
            fillRect(a,y,b-a+1,1,c);
        }
    }
}

// Convierte la matriz 5x7 de una letra en pequeños rectángulos de píxeles.
void AdafruitGFX_Pico::dibujarCaracter(uint8_t codigo,int16_t x,int16_t y){
    if(x>=ancho_||y>=alto_||x+6*tam_<=0||y+8*tam_<=0)return;
    const uint8_t *bits=FUENTE_RETRO[codigo];
    for(int row=0;row<7;++row){
        int run=-1;
        for(int col=0;col<=5;++col){
            const bool on=(col<5) && ((bits[row]>>(4-col))&1);
            if(on && run<0)run=col;
            if(!on && run>=0){fillRect(x+run*tam_,y+row*tam_,(col-run)*tam_,tam_,textoColor_);run=-1;}
        }
    }
}
// Escribe el texto carácter a carácter; también respeta salto de línea y salto automático.
std::size_t AdafruitGFX_Pico::print(const char *txt){
    if(!txt)return 0;
    std::size_t count=0;
    for(const uint8_t *p=reinterpret_cast<const uint8_t*>(txt);*p;++p){
        if(*p=='\n'){cursorX_=0;cursorY_+=8*tam_;continue;}
        if(wrap_ && cursorX_+6*tam_>ancho_){cursorX_=0;cursorY_+=8*tam_;}
        dibujarCaracter(*p,cursorX_,cursorY_);cursorX_+=6*tam_;++count;
    }
    return count;
}
std::size_t AdafruitGFX_Pico::print(int valor){char buffer[20];std::snprintf(buffer,sizeof(buffer),"%d",valor);return print(buffer);}
// Calcula el espacio de un texto para poder centrarlo en la pantalla.
void AdafruitGFX_Pico::getTextBounds(const char *txt,int16_t x,int16_t y,
                                     int16_t *x1,int16_t *y1,uint16_t *w,uint16_t *h) const {
    if(x1)*x1=x;
    if(y1)*y1=y;
    uint16_t cantidad=0;
    if(txt)for(const char *p=txt;*p;p++)if(*p!='\n')++cantidad;
    if(w)*w=cantidad*6*tam_;
    if(h)*h=8*tam_;
}

// Mide un texto, calcula su posición horizontal y lo dibuja centrado.
void PantallaUI::textoCentrado(const char *txt, int y, uint8_t tam, uint16_t color) {
    int16_t x1, y1;
    uint16_t w, h;
    _tft.setTextSize(tam);
    _tft.setTextColor(color);

// Dibuja dos bordes paralelos para dar profundidad a los cuadros de diálogo.
void PantallaUI::marcoDoble(int x, int y, int w, int h, uint16_t color, int radio) {
    if (radio > 0) {
        _tft.drawRoundRect(x, y, w, h, radio, color);
        _tft.drawRoundRect(x + 1, y + 1, w - 2, h - 2, radio, color);
    } else {
        _tft.drawRect(x, y, w, h, color);
        _tft.drawRect(x + 1, y + 1, w - 2, h - 2, color);
    }
}

// Superpone dos líneas casi iguales para dar mayor grosor al personaje.
void PantallaUI::lineaGruesa(int x0, int y0, int x1, int y1, uint16_t color) {
    _tft.drawLine(x0, y0, x1, y1, color);
    _tft.drawLine(x0 + 1, y0, x1 + 1, y1, color);
}

// Calcula la coordenada vertical de un vehículo y evita índices fuera de los carriles.
int PantallaUI::yCarril(int carril) {
    if (carril < 0) carril = 0;
    if (carril >= Carretera::NUM_CARRILES) carril = Carretera::NUM_CARRILES - 1;
    return Carretera::CARRIL_Y0 + carril * Carretera::CARRIL_ALTO +
           (Carretera::CARRIL_ALTO - Sprite::AUTO_H) / 2;
}

// ---- Slide 3: Juego - Laberinto -------------------------------------------
// Pantalla 3: dibuja los límites y paredes; la micro es la meta.
void PantallaUI::slideLaberinto() {
    // Marco exterior (grosor 2)
    _tft.drawRect(MAZE_X, MAZE_Y, MAZE_W, MAZE_H, BLANCO);
    _tft.drawRect(MAZE_X + 1, MAZE_Y + 1, MAZE_W - 2, MAZE_H - 2, BLANCO);

    // Paredes interiores
    const size_t n = sizeof(MUROS) / sizeof(MUROS[0]);
    for (size_t i = 0; i < n; i++) {
        const int x1 = escX(MUROS[i].x1), y1 = escY(MUROS[i].y1);
        const int x2 = escX(MUROS[i].x2), y2 = escY(MUROS[i].y2);
        if (y1 == y2) {                       // horizontal
            _tft.fillRect(std::min(x1, x2), y1, abs(x2 - x1) + GROSOR, GROSOR, BLANCO);
        } else {                              // vertical
            _tft.fillRect(x1, std::min(y1, y2), GROSOR, abs(y2 - y1) + GROSOR, BLANCO);
        }
    }

    // Punto de partida (arriba-izquierda) y meta (abajo-derecha)
    // El estudiante movil se dibuja desde main.cpp segun las perillas.
    dibujarAutobus(MAZE_X + MAZE_W - Sprite::AUTOBUS_W - 8,
                   MAZE_Y + MAZE_H - Sprite::AUTOBUS_H - 8);
}

// ---- Slide 4: Juego - Carretera -------------------------------------------
// Pantalla 4: prepara tres carriles y sus líneas divisorias.
void PantallaUI::slideCarretera() {
    using namespace Pantalla;
    // Borde de la calle
    _tft.drawRect(JUEGO_X, JUEGO_Y, JUEGO_W, JUEGO_H, BLANCO);

    // Líneas divisorias de carriles (discontinuas)
    for (int c = 1; c < Carretera::NUM_CARRILES; c++) {
        const int y = Carretera::CARRIL_Y0 + c * Carretera::CARRIL_ALTO;
        for (int x = JUEGO_X + 4; x < JUEGO_X + JUEGO_W - 16; x += 22) {
            _tft.drawFastHLine(x, y, 14, BLANCO);
            _tft.drawFastHLine(x, y + 1, 14, BLANCO);
        }
    }
}

// Collision AABB de estudiante contra muros geométricos del boceto original.
// Comprueba si la caja del estudiante cruza el borde o toca alguna pared del laberinto.
bool PantallaUI::colisionLaberinto(int x, int y, int w, int h) const {
    if (x < MAZE_X + 2 || y < MAZE_Y + 2 ||
        x + w > MAZE_X + MAZE_W - 2 ||
        y + h > MAZE_Y + MAZE_H - 2) return true;

    for (const Muro &muro : MUROS) {
        int x0 = std::min(escX(muro.x1), escX(muro.x2));
        int x1 = std::max(escX(muro.x1), escX(muro.x2)) + GROSOR;
        int y0 = std::min(escY(muro.y1), escY(muro.y2));
        int y1 = std::max(escY(muro.y1), escY(muro.y2)) + GROSOR;
        if (x < x1 && x+w > x0 && y < y1 && y+h > y0) return true;
    }
    return false;
}

// 11 · CLASE Perilla (reutilizada del proyecto anterior)
// Convierte la señal analógica del potenciómetro en movimientos discretos.
class Perilla {
public:
    explicit Perilla(unsigned int pin);
    void iniciar() const;
    uint16_t leer() const; // ADC 0 a 4095.
    int direccion() const; // -1, 0, +1: giro relativo a la zona central.
private:
    unsigned int pin_;
};

// ========================================================================
// Perilla: implementacion
// ========================================================================

// Al crear cada objeto Perilla se guarda su GPIO de lectura.
Perilla::Perilla(unsigned int pin) : pin_(pin) {}
// Habilita el pin para que el conversor ADC pueda leer la tensión.
void Perilla::iniciar() const { adc_gpio_init(pin_); }
// Selecciona ADC0 para GP26 o ADC1 para GP27 y devuelve el valor de 12 bits (0..4095).
uint16_t Perilla::leer() const {
    adc_select_input(pin_ - 26); // GPIO26 => ADC0; GPIO27 => ADC1.
    return adc_read();
}
// Usa una zona neutra central: devuelve -1, 0 o 1 según la posición del potenciómetro.
int Perilla::direccion() const {
    const uint16_t valor = leer();
    if(valor < 1660) return -1;
    if(valor > 2450) return 1;
    return 0;
}

// 12 · CLASE Boton2 (recibida de otro grupo y adaptada)
// Su máquina de estados permite reconocer pulsaciones sin bloquear el programa.
enum class EstadoBoton {
    REPOSO,
    PRESIONADO,
    ESPERANDO_PRESION
};

// Se definen clic sencillo, doble, triple y dos duraciones de pulsación larga.
enum class Gesto {
    NADA,
    PULSADO1VEZ,
    PULSADO2VECES,
    PULSADO3VECES,
    PULSACIONLARGA,
    PULSACIONMUYLARGA
};

// El constructor prepara un GPIO; leer() analiza cambios; leerGesto() entrega el evento.
class Boton2 {
public:
    // El número es GPIO, no número físico de la placa.
    explicit Boton2(int pin);
    void leer();
    Gesto leerGesto();

    // Parámetros ajustables para el proyecto que reutiliza esta clase.
    uint32_t tiempoLargo = 1000;
    uint32_t tiempoMuyLargo = 5000;
    uint32_t tiempoEntrePresionado = 400;
    uint32_t tiempoAntirrebote = 25;

private:
    int pin;
    EstadoBoton estado = EstadoBoton::REPOSO;
    Gesto gestoPendiente = Gesto::NADA;
    bool presionado = false;
    int numeroPresiones = 0;
    uint32_t instanteCambio = 0;
    uint32_t instanteUltimaLiberacion = 0;
    bool lecturaBrutaAnterior = false;
    uint32_t instanteCambioBruto = 0;
};

// ========================================================================
// Boton2: implementacion
// ========================================================================

// conserva sus gestos y su API, añade antirrebote y tiempos de liberación
// independientes para identificar pulsaciones múltiples correctamente.
// Configura el pin como entrada con pull-up; un botón presionado entrega nivel bajo.
Boton2::Boton2(int pin) : pin(pin) {
    gpio_init(pin);
    gpio_set_dir(pin, GPIO_IN);
    gpio_pull_up(pin); // Cablear entre GPIO y GND: pulsado = nivel 0.
    presionado = !gpio_get(pin);
    lecturaBrutaAnterior = presionado;
    instanteCambioBruto = to_ms_since_boot(get_absolute_time());
}

// Actualiza el estado del botón en cada vuelta del juego, sin esperas bloqueantes.
void Boton2::leer() {
    const uint32_t ahora = to_ms_since_boot(get_absolute_time());
    // Se invierte la lectura porque el pulsador conecta el GPIO a GND.
    const bool bruto = !gpio_get(pin);

    // Guarda el momento del cambio eléctrico bruto para filtrar los rebotes del contacto.
    if (bruto != lecturaBrutaAnterior) {
        lecturaBrutaAnterior = bruto;
        instanteCambioBruto = ahora;
    }

    bool apretaron = false;
    bool soltaron = false;
    // Solo se acepta el cambio si se mantuvo estable por tiempoAntirrebote.
    if (bruto != presionado &&
        static_cast<uint32_t>(ahora - instanteCambioBruto) >= tiempoAntirrebote) {
        apretaron = !presionado && bruto;
        soltaron = presionado && !bruto;
        presionado = bruto;
    }

    switch (estado) {
        // REPOSO: espera el inicio de una pulsación.
        case EstadoBoton::REPOSO:
            if (apretaron) {
                numeroPresiones = 0;
                instanteCambio = ahora;
                estado = EstadoBoton::PRESIONADO;
            }
            break;

        // PRESIONADO: mide cuánto duró y decide si fue corta, larga o muy larga.
        case EstadoBoton::PRESIONADO:
            if (soltaron) {
                const uint32_t duracion = ahora - instanteCambio;
                if (duracion >= tiempoMuyLargo) {
                    gestoPendiente = Gesto::PULSACIONMUYLARGA;
                    numeroPresiones = 0;
                    estado = EstadoBoton::REPOSO;
                } else if (duracion >= tiempoLargo) {
                    gestoPendiente = Gesto::PULSACIONLARGA;
                    numeroPresiones = 0;
                    estado = EstadoBoton::REPOSO;
                } else {
                    ++numeroPresiones;
                    instanteUltimaLiberacion = ahora;
                    estado = EstadoBoton::ESPERANDO_PRESION;
                }
            } else if (presionado &&
                       static_cast<uint32_t>(ahora - instanteCambio) >= tiempoMuyLargo) {
                gestoPendiente = Gesto::PULSACIONMUYLARGA;
                numeroPresiones = 0;
                estado = EstadoBoton::REPOSO;
            }
            break;

        // ESPERANDO_PRESION: espera un segundo/tercer clic durante la ventana configurada.
        case EstadoBoton::ESPERANDO_PRESION:
            if (apretaron) {
                instanteCambio = ahora;
                estado = EstadoBoton::PRESIONADO;
            } else if (static_cast<uint32_t>(ahora - instanteUltimaLiberacion) >=
                       tiempoEntrePresionado) {
                switch (numeroPresiones) {
                    case 1: gestoPendiente = Gesto::PULSADO1VEZ; break;
                    case 2: gestoPendiente = Gesto::PULSADO2VECES; break;
                    case 3: gestoPendiente = Gesto::PULSADO3VECES; break;
                    default: break;
                }
                numeroPresiones = 0;
                estado = EstadoBoton::REPOSO;
            }
            break;
    }
}

// Devuelve el gesto acumulado y lo borra para que no se ejecute dos veces.
Gesto Boton2::leerGesto() {
    const Gesto gesto = gestoPendiente;
    gestoPendiente = Gesto::NADA;
    return gesto;
}

// Detecta superposición de rectángulos (colisión AABB) entre personajes, autos y metas.

    return x0 < x1+w1 && x0+w0 > x1 && y0 < y1+h1 && y0+h0 > y1;

// 19 · REGLAS DEL BOTÓN DE CONTINUAR
// En pantallas informativas avanza; en niveles exige victoria o permite reintentar una derrota.
void controlarBotonContinuar() {
    if (!esNivel(slideActual)) {
        iniciarSlide((slideActual+1) % PantallaUI::TOTAL_SLIDES);
        return;
    }
    if (estado == Estado::Superado) {
        iniciarSlide((slideActual+1) % PantallaUI::TOTAL_SLIDES);
    } else if (estado == Estado::Fallido) {
        iniciarSlide(slideActual);
    } else {
        // En curso: no se permite saltar el desafio.
        printf("Desafio pendiente: completa el nivel para continuar.\n");
    }
}

// 20 · ACTUALIZACIÓN DE ANIMACIÓN

    if(slideActual == PantallaUI::SLIDE_CARRETERA) {
        tft.fillRect(antesX,antesY,Sprite::AUTOBUS_W,Sprite::AUTOBUS_H,Colores::NEGRO);

    } else {
        tft.fillRect(antesX,antesY,Sprite::ESTUDIANTE_W,Sprite::ESTUDIANTE_H,Colores::NEGRO);
    }
    dibujarMoviles();
}

// 21 · LÓGICA PRINCIPAL DE LOS DESAFÍOS
// Comprueba tiempo, lee dirección de perillas y resuelve movimiento, colisiones y metas.
void actualizarNivel(uint32_t ahoraMs) {
    if (!esNivel(slideActual) || estado != Estado::EnCurso) return;
    const int limite = limiteActual();
    // Los niveles cronometrados muestran el tiempo restante; la sala se juega sin límite.
    if (limite > 0) {
        const int transcurrido = static_cast<int>((ahoraMs-inicioNivelMs)/1000);
        const int restante = std::max(0,limite-transcurrido);
        if(restante != segundosPrevios) {
            segundosPrevios=restante;
            pantalla.dibujarTiempo(restante);
        }
        // El LED de GP7 parpadea en los últimos cinco segundos.
        if(restante <= 5) {
            gpio_put(PIN_LED, (ahoraMs / 250)%2 == 0);
        } else gpio_put(PIN_LED,0);
        if(restante == 0){cambiarEstado(Estado::Fallido);return;}
    } else gpio_put(PIN_LED,0);

    // La carretera utiliza una actualización más frecuente que las demás etapas.
    const uint32_t intervalo = (slideActual == PantallaUI::SLIDE_CARRETERA)
                               ? INTERVALO_CARRETERA_MS : INTERVALO_DIBUJO_MS;
    if(ahoraMs-ultimoDibujoMs < intervalo)return;
    ultimoDibujoMs=ahoraMs;

    const int antesX=jugadorX,antesY=jugadorY;
    // Se convierte cada posición de la perilla en desplazamientos horizontal y vertical.
    const int dx=perillaHorizontal.direccion() *
                 (slideActual == PantallaUI::SLIDE_CARRETERA ? PASO_MICRO : 6);
    const int dy=perillaVertical.direccion() *
                 (slideActual == PantallaUI::SLIDE_CARRETERA ? PASO_MICRO_VERTICAL : 6);

    // DESAFÍO 1 · LABERINTO
    // Mueve al estudiante sin atravesar paredes; llegar a la micro marca victoria.
    if(slideActual == PantallaUI::SLIDE_LABERINTO) {
        // Movimiento pixel a pixel: evita atravesar muros delgados (2 px).
        for(int paso=0;paso<std::abs(dx);++paso) {
            int proximoX=jugadorX+(dx>0 ? 1 : -1);
            if(pantalla.colisionLaberinto(proximoX,jugadorY))break;
            jugadorX=proximoX;
        }
        for(int paso=0;paso<std::abs(dy);++paso) {
            int proximoY=jugadorY+(dy>0 ? 1 : -1);
            if(pantalla.colisionLaberinto(jugadorX,proximoY))break;
            jugadorY=proximoY;
        }
        if(interseccion(jugadorX,jugadorY,Sprite::ESTUDIANTE_W,Sprite::ESTUDIANTE_H,
                        266,204,Sprite::AUTOBUS_W,Sprite::AUTOBUS_H)){
            dibujarMovimiento(antesX,antesY);
            cambiarEstado(Estado::Superado);return;
        }
    // DESAFÍO 2 · CARRETERA
    // Mueve la micro, desplaza autos en sentido contrario y comprueba choques o meta.
    } else if(slideActual == PantallaUI::SLIDE_CARRETERA) {
        jugadorX=std::clamp(jugadorX+dx,10,268);
        jugadorY=std::clamp(jugadorY+dy,24,211);
        for(auto &v:autos){
            v.x-=v.paso;
            if(v.x < -Sprite::AUTO_W) v.x = 326 + (v.paso*10);
            if(interseccion(jugadorX,jugadorY,Sprite::AUTOBUS_W,Sprite::AUTOBUS_H,
                            v.x,v.y,Sprite::AUTO_W,Sprite::AUTO_H)){
                // Redibujar despues de colision antes del mensaje.
                pantalla.dibujarSlide(slideActual);
                dibujarMetaCarretera(true);
                dibujarMoviles();
                cambiarEstado(Estado::Fallido);
                return;
            }
        }
        // La micro completa la etapa cuando su borde derecho toca la META.
        if(jugadorX >= 268){
            pantalla.dibujarSlide(slideActual);
            dibujarMetaCarretera(true);
            dibujarMoviles();
            cambiarEstado(Estado::Superado);return;
        }
    // DESAFÍO 3 · SALA DE CLASES
    // Mueve al estudiante hasta la puerta 304, sin cronómetro.
    } else if(slideActual == PantallaUI::SLIDE_SALA) {
        jugadorX=std::clamp(jugadorX+dx,9,291);
        jugadorY=std::clamp(jugadorY+dy,24,209);
        if(interseccion(jugadorX,jugadorY,Sprite::ESTUDIANTE_W,Sprite::ESTUDIANTE_H,
                        266,174,40,52)){
            dibujarMovimiento(antesX,antesY);
            cambiarEstado(Estado::Superado);return;
        }
    }

    if(antesX!=jugadorX || antesY!=jugadorY || slideActual==PantallaUI::SLIDE_CARRETERA)
        dibujarMovimiento(antesX,antesY);
}

// Inicia la Pico 2 W, ADC, pantalla, controles y LED; no inicializa audio.
int main() {
    // Habilita la comunicación estándar (USB si está activada en CMake).
    stdio_init_all();
    // Inicializa el conversor analógico-digital para ambas perillas.
    adc_init();
    perillaVertical.iniciar();
    perillaHorizontal.iniciar();
    // Boton2 configura cada GPIO en el constructor.
    // Prepara el LED como salida y lo mantiene apagado al arrancar.
    gpio_init(PIN_LED);
    gpio_set_dir(PIN_LED, GPIO_OUT);
    gpio_put(PIN_LED,0);
    // Sin inicializar PWM ni el pin GP13 (parlante desconectado).
    // Inicia el controlador ST7789V y dibuja la pantalla de bienvenida.
    pantalla.begin();
    iniciarSlide(PantallaUI::SLIDE_INICIO);
    printf("DORE 8 sin audio | GP10 reservado | GP11 continuar | GP26 vertical | GP27 horizontal\n");

    // BUCLE PRINCIPAL
    // Cada iteración lee botones, comprueba gestos, actualiza el nivel y espera 5 ms.
    while(true) {
        const uint32_t ahora=relojMs();
        // Sin audio.
        // Se lee GP10 pero sus gestos se descartan porque el audio quedó desactivado.
        botonMusica.leer();
        botonAvanzar.leer();
        // Para DORE 8, solo un clic simple realiza acciones.
        // Los gestos doble, triple y largo están disponibles para usos futuros.
        // Botón GP10 reservado hasta instalar un amplificador de audio.
        (void)botonMusica.leerGesto();
        // Solo el clic sencillo en GP11 puede comenzar, avanzar o reiniciar una etapa.
        if (botonAvanzar.leerGesto() == Gesto::PULSADO1VEZ)
            controlarBotonContinuar();
        // Actualiza cronómetro, posiciones, colisiones, LED y dibujo de la etapa.
        actualizarNivel(ahora);
        // Una espera breve limita el sondeo sin detener de forma visible el juego.
        sleep_ms(5);
    }
}
